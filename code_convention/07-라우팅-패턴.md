## React Router 7 개요

### 주요 특징

- **Flat File Routing**: 파일명 기반 자동 라우트 생성
- **Type Safety**: 자동 타입 생성 (loader, action, params)
- **Data Loading**: loader/action 패턴
- **Nested Layouts**: 레이아웃 라우트
- **Error Boundaries**: 라우트별 에러 처리

### 설정

`@react-router/fs-routes`를 사용하면 파일 시스템 기반 자동 라우팅이 적용됩니다. 별도의 라우트 설정 파일이 필요 없습니다.

```typescript
// react-router.config.ts
import type { Config } from '@react-router/dev/config';

export default {
  // File-based routing 활성화 (기본값)
  appDirectory: 'app',
} satisfies Config;
```

```typescript
// app/routes.ts
import { type RouteConfig } from '@react-router/dev/routes';
import { flatRoutes } from '@react-router/fs-routes';

export default flatRoutes() satisfies RouteConfig;
```

> **Note**: `flatRoutes()`가 `app/routes/` 디렉토리의 파일 구조를 자동으로 라우트로 변환합니다.

## 파일 라우팅 규칙

### 기본 규칙

| 파일명 | URL | 설명 |
|--------|-----|------|
| `_index.tsx` | `/` | 인덱스 라우트 |
| `about.tsx` | `/about` | 기본 라우트 |
| `about_.contact.tsx` | `/about/contact` | 중첩 (레이아웃 없음) |
| `projects.tsx` | `/projects/*` | 레이아웃 라우트 |
| `projects._index.tsx` | `/projects` | 프로젝트 인덱스 |
| `projects.$id.tsx` | `/projects/:id` | 동적 파라미터 |
| `projects.$id_.edit.tsx` | `/projects/:id/edit` | 레이아웃 탈출 |
| `$.tsx` | `/*` | Splat (Catch-all) |

### 특수 파일명

```
routes/
├── _index.tsx              # 루트 인덱스 (/)
├── _layout.tsx             # 루트 레이아웃 (선택적)
│
├── about.tsx               # /about
├── about_.team.tsx         # /about/team (about 레이아웃 무시)
│
├── blog.tsx                # /blog 레이아웃
├── blog._index.tsx         # /blog
├── blog.$slug.tsx          # /blog/:slug
│
├── _auth.tsx               # 인증 레이아웃 (URL 미포함)
├── _auth.login.tsx         # /login
├── _auth.register.tsx      # /register
│
├── api.$.tsx               # /api/* (API 라우트)
├── [sitemap.xml].tsx       # /sitemap.xml (특수문자 이스케이프)
│
└── $.tsx                   # /* (404 Catch-all)
```

### 동적 라우트

```tsx
// routes/projects.$id.tsx
// URL: /projects/123 -> params.id = "123"

// routes/blog.$category.$slug.tsx
// URL: /blog/tech/hello-world -> params.category = "tech", params.slug = "hello-world"

// routes/files.$.tsx (Splat)
// URL: /files/images/photo.jpg -> params["*"] = "images/photo.jpg"

// routes/[sitemap.xml].tsx (Escaped)
// URL: /sitemap.xml
```

## 라우트 파일 구조

### 기본 구조

```tsx
// routes/projects.$id.tsx
import type { Route } from './+types/projects.$id';

// 메타데이터 (SEO)
export function meta({ data }: Route.MetaArgs) {
  return [
    { title: data?.project?.title ?? 'Project' },
    { name: 'description', content: data?.project?.description },
    { property: 'og:title', content: data?.project?.title },
  ];
}

// 링크 (CSS, 프리로드 등)
export function links() {
  return [
    { rel: 'stylesheet', href: '/styles/project.css' },
    { rel: 'preload', href: '/images/hero.jpg', as: 'image' },
  ];
}

// 헤더 (HTTP 헤더)
export function headers() {
  return {
    'Cache-Control': 'public, max-age=3600',
  };
}

// 로더 (서버 데이터)
export async function loader({ params, request }: Route.LoaderArgs) {
  const project = await getProject(params.id);

  if (!project) {
    throw new Response('Not Found', { status: 404 });
  }

  return { project };
}

// 액션 (폼 제출)
export async function action({ params, request }: Route.ActionArgs) {
  const formData = await request.formData();
  const intent = formData.get('intent');

  if (intent === 'delete') {
    await deleteProject(params.id);
    return redirect('/projects');
  }

  return { error: 'Unknown action' };
}

// 컴포넌트
export default function ProjectPage({ loaderData }: Route.ComponentProps) {
  const { project } = loaderData;

  return (
    <div>
      <h1>{project.title}</h1>
      <p>{project.description}</p>
    </div>
  );
}

// 에러 바운더리
export function ErrorBoundary({ error }: Route.ErrorBoundaryProps) {
  if (error instanceof Response && error.status === 404) {
    return <NotFoundPage />;
  }

  return (
    <div>
      <h1>오류가 발생했습니다</h1>
      <p>{error.message}</p>
    </div>
  );
}

// 로딩 UI (HydrateFallback)
export function HydrateFallback() {
  return <ProjectSkeleton />;
}
```

### 타입 안전성

```tsx
// React Router 7은 자동으로 타입 생성
// routes/+types/projects.$id.ts (자동 생성)

import type { Route } from './+types/projects.$id';

// params 타입 자동 추론

  params.id; // string (타입 안전)
}

// loaderData 타입 자동 추론
export default function Page({ loaderData }: Route.ComponentProps) {
  loaderData.project; // loader 반환 타입 기반
```

## 데이터 로딩

### Loader 패턴

```tsx
// routes/projects._index.tsx
import type { Route } from './+types/projects._index';
import { fetchProjects } from '@/features/projects';

export async function loader({ request }: Route.LoaderArgs) {
  const url = new URL(request.url);
  const page = parseInt(url.searchParams.get('page') ?? '1');
  const category = url.searchParams.get('category') ?? undefined;

  const projects = await fetchProjects({ page, category });

  return { projects };
}

export default function ProjectsPage({ loaderData }: Route.ComponentProps) {
  const { projects } = loaderData;

  return <ProjectList projects={projects} />;
}
```

### Action 패턴

```tsx
// routes/projects.new.tsx
import type { Route } from './+types/projects.new';
import { createProject } from '@/features/projects';
import { projectSchema } from '@/features/projects/schemas';
import { redirect } from 'react-router';

export async function action({ request }: Route.ActionArgs) {
  const formData = await request.formData();
  const data = Object.fromEntries(formData);

  // 유효성 검사
  const result = projectSchema.safeParse(data);
  if (!result.success) {
    return {
      errors: result.error.flatten().fieldErrors,
    };
  }

  // 생성
  const project = await createProject(result.data);

  return redirect(`/projects/${project.id}`);
}

export default function NewProjectPage({ actionData }: Route.ComponentProps) {
  return (
    <Form method="post">
      <input name="title" />
      {actionData?.errors?.title && (
        <span>{actionData.errors.title[0]}</span>
      )}
      <button type="submit">생성</button>
    </Form>
  );
}
```

### TanStack Query와 함께 사용

```tsx
// routes/projects.$id.tsx
import { queryClient } from '@/lib/query-client';
import { projectKeys, fetchProject } from '@/features/projects';

// 프리페치
export async function loader({ params }: Route.LoaderArgs) {
  await queryClient.prefetchQuery({
    queryKey: projectKeys.detail(params.id),
    queryFn: () => fetchProject(params.id),
  });

  return null;
}

// 컴포넌트에서 useQuery 사용
export default function ProjectPage() {
  const { id } = useParams();
  const { data: project } = useProject(id!);

  return <ProjectDetail project={project} />;
}
```

## 보호된 라우트

### 인증 필요 레이아웃

```tsx
// routes/_protected.tsx
import { Outlet, redirect } from 'react-router';
import type { Route } from './+types/_protected';
import { getSession } from '@/lib/session.server';

export async function loader({ request }: Route.LoaderArgs) {
  const session = await getSession(request);

  if (!session?.user) {
    const url = new URL(request.url);
    const redirectTo = url.pathname + url.search;
    return redirect(`/login?redirectTo=${encodeURIComponent(redirectTo)}`);
  }

  return { user: session.user };
}

export default function ProtectedLayout({ loaderData }: Route.ComponentProps) {
  return (
    <div>
      <Header user={loaderData.user} />
      <Outlet />
    </div>
  );
}
```

### 클라이언트 사이드 보호

```tsx
// shared/components/auth/RequireAuth.tsx
import { Navigate, useLocation } from 'react-router';
import { useAuthStore } from '@/features/auth';

type RequireAuthProps = {
  children: React.ReactNode;
  redirectTo?: string;
};

export function RequireAuth({
  children,
  redirectTo = '/login',
}: RequireAuthProps) {
  const location = useLocation();
  const { isAuthenticated, isLoading } = useAuthStore();

  if (isLoading) {
    return <LoadingScreen />;
  }

  if (!isAuthenticated) {
    return (
      <Navigate
        to={`${redirectTo}?redirectTo=${encodeURIComponent(location.pathname)}`}
        replace
      />
    );
  }

  return <>{children}</>;
}

// 사용
// routes/_protected.tsx
export default function ProtectedLayout() {
  return (
    <RequireAuth>
      <Outlet />
    </RequireAuth>
  );
}
```

### 역할 기반 접근 제어

```tsx
// shared/components/auth/RequireRole.tsx
import { Navigate } from 'react-router';
import { useAuthStore } from '@/features/auth';

type Role = 'admin' | 'member' | 'guest';

type RequireRoleProps = {
  children: React.ReactNode;
  roles: Role[];
  fallback?: React.ReactNode;
};

export function RequireRole({
  children,
  roles,
  fallback = <AccessDenied />,
}: RequireRoleProps) {
  const user = useAuthStore((state) => state.user);

  if (!user || !roles.includes(user.role)) {
    return <>{fallback}</>;
  }

  return <>{children}</>;
}

// 사용
function AdminPanel() {
  return (
    <RequireRole roles={['admin']}>
      <AdminDashboard />
    </RequireRole>
  );
}
```

```tsx
// routes/_auth.login.tsx
import { redirect } from 'react-router';

export async function action({ request }: Route.ActionArgs) {
  const formData = await request.formData();
  const url = new URL(request.url);
  const redirectTo = url.searchParams.get('redirectTo') ?? '/dashboard';

  const result = await login(formData);

  if (result.success) {
    return redirect(redirectTo);
  }

  return { error: result.error };
}

export default function LoginPage() {
  const [searchParams] = useSearchParams();
  const redirectTo = searchParams.get('redirectTo');

  return (

      <input name="email" type="email" />
      <input name="password" type="password" />
      <input type="hidden" name="redirectTo" value={redirectTo ?? ''} />

    </Form>
  );
}
```
