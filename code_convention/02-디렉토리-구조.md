**root.tsx 예시:**
```tsx
import { Outlet, Scripts, ScrollRestoration } from 'react-router';
import { Providers } from './providers';

export default function Root() {

    <html lang="ko">
      <head>

        <Providers>
          <Outlet />
        </Providers>
        <ScrollRestoration />
        <Scripts />
      </body>
    </html>
  );
}
```

## Routes 구조

### Flat File Routing 규칙

| 파일명 패턴 | URL | 설명 |
|-------------|-----|------|
| `_index.tsx` | `/` | 인덱스 라우트 |
| `about.tsx` | `/about` | 기본 라우트 |
| `blog.tsx` | `/blog/*` | 레이아웃 라우트 |
| `blog._index.tsx` | `/blog` | 블로그 인덱스 |
| `blog.$slug.tsx` | `/blog/:slug` | 동적 라우트 |
| `projects.$id.edit.tsx` | `/projects/:id/edit` | 중첩 라우트 |
| `_auth.tsx` | - | 패스리스 레이아웃 |
| `_auth.login.tsx` | `/login` | 레이아웃 하위 라우트 |
| `$.tsx` | `/*` | Splat (Catch-all) |
| `[sitemap.xml].tsx` | `/sitemap.xml` | 이스케이프 라우트 |

### 라우트 파일 구조

```tsx
// routes/projects.$id.tsx
import type { Route } from './+types/projects.$id';
import { ProjectDetail } from '~/features/projects';

// 메타데이터
export function meta({ data }: Route.MetaArgs) {
  return [
    { title: data?.project.title ?? 'Project' },
    { name: 'description', content: data?.project.description },
  ];
}

// 로더 (서버 데이터 페칭)
export async function loader({ params }: Route.LoaderArgs) {
  const project = await getProject(params.id);
  if (!project) throw new Response('Not Found', { status: 404 });
  return { project };
}

// 컴포넌트
export default function ProjectPage({ loaderData }: Route.ComponentProps) {
  return <ProjectDetail project={loaderData.project} />;
}

// 에러 바운더리
export function ErrorBoundary({ error }: Route.ErrorBoundaryProps) {
  return <div>Error: {error.message}</div>;
}
```

### 모듈 간 의존성 규칙

```
┌─────────────────────────────────────────┐
│                 routes/                 │  ← 페이지 레벨
└─────────────────────────────────────────┘
                    │
                    ▼
┌─────────────────────────────────────────┐
│               features/                 │  ← 도메인 레벨
│  ┌─────────┐  ┌─────────┐  ┌─────────┐  │
│  │  auth   │  │ projects│  │  blog   │  │
│  └─────────┘  └─────────┘  └─────────┘  │
└─────────────────────────────────────────┘
                    │
                    ▼
┌─────────────────────────────────────────┐
│                shared/                  │  ← 공통 레벨
│  components, hooks, utils, types, lib   │
└─────────────────────────────────────────┘
```

**규칙:**
- `routes/` → `features/` 및 `shared/` 사용 가능
- `features/` → `shared/`만 사용 가능 (다른 feature 직접 참조 금지)
- `shared/` → 외부 라이브러리만 사용 가능

### 어디에 둘 것인가?

```
┌─────────────────────────────────────────────────────────────┐
│ 이 코드가 특정 기능(도메인)에 종속되어 있는가?               │
│                                                             │
│   YES ─────────────────────────► features/{feature}/        │
│    │                                                        │
│   NO                                                        │
│    │                                                        │
│    ▼                                                        │
│ 이 코드가 여러 곳에서 재사용되는가?                          │
│                                                             │
│   YES ─────────────────────────► shared/                    │
│    │                                                        │
│   NO                                                        │
│    │                                                        │
│    ▼                                                        │
│ 이 코드가 라우트/페이지에만 사용되는가?                      │
│                                                             │
│   YES ─────────────────────────► routes/ (해당 라우트 파일) │
│    │                                                        │
│   NO                                                        │
│    │                                                        │
│    ▼                                                        │
│ 앱 전체 설정/초기화 관련인가?                                │
│                                                             │
│   YES ─────────────────────────► app/                       │
└─────────────────────────────────────────────────────────────┘
```
