## React 최적화

### React 19 Compiler - 자동 메모이제이션

> **중요**: React 19부터 React Compiler가 자동으로 메모이제이션을 처리합니다.
> `memo`, `useMemo`, `useCallback`은 **사용하지 않습니다**.

```tsx
// ❌ React 19에서 불필요
import { memo, useMemo, useCallback } from 'react';

const ProjectCard = memo(function ProjectCard() { ... });
const filtered = useMemo(() => items.filter(...), [items]);
const handleClick = useCallback(() => {}, []);

// ✅ React 19 - 그냥 작성하면 Compiler가 최적화
function ProjectCard({ project }: { project: Project }) {
  return (
    <Card>
      <CardHeader>
        <CardTitle>{project.title}</CardTitle>
      </CardHeader>
    </Card>
  );
}

function ProjectList({ projects, filter }: Props) {
  // Compiler가 자동으로 메모이제이션
  const filteredProjects = projects.filter((p) => {
    if (!filter) return true;
    return p.category === filter;
  });

  return (
    <div>
      {filteredProjects.map((project) => (
        <ProjectCard key={project.id} project={project} />
      ))}
    </div>
  );
}

function ParentComponent() {
  const [count, setCount] = useState(0);

  // Compiler가 자동으로 최적화
  const handleClick = () => {
    console.log('clicked');
  };

  return <ChildComponent onClick={handleClick} />;
}
```

```tsx
// Intersection Observer 기반
function LazyImage({ src, alt, ...props }: ImageProps) {
  const [isLoaded, setIsLoaded] = useState(false);
  const imgRef = useRef<HTMLImageElement>(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsLoaded(true);
          observer.disconnect();
        }
      },
      { rootMargin: '100px' }
    );

    if (imgRef.current) {
      observer.observe(imgRef.current);
    }

    return () => observer.disconnect();
  }, []);

  return (
    <img
      ref={imgRef}
      src={isLoaded ? src : '/placeholder.jpg'}
      alt={alt}
      {...props}
    />
  );
}
```

### 이미지 컴포넌트

```tsx
// shared/components/OptimizedImage.tsx
import { useState } from 'react';
import { cn } from '@/utils/cn';

type OptimizedImageProps = {
  src: string;
  alt: string;
  width: number;
  height: number;
  className?: string;
  priority?: boolean;
};

export function OptimizedImage({
  src,
  alt,
  width,
  height,
  className,
  priority = false,
}: OptimizedImageProps) {
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState(false);

  // WebP 버전 생성 (실제로는 빌드 시 또는 CDN에서 처리)
  const webpSrc = src.replace(/\.(jpg|png)$/, '.webp');

  if (error) {
    return (
      <div
        className={cn(
          'flex items-center justify-center bg-muted',
          className
        )}
        style={{ width, height }}
      >
        <span className="text-muted-foreground">이미지 로드 실패</span>
      </div>
    );
  }

  return (
    <picture>
      <source srcSet={webpSrc} type="image/webp" />
      <img
        src={src}
        alt={alt}
        width={width}
        height={height}
        loading={priority ? 'eager' : 'lazy'}
        decoding={priority ? 'sync' : 'async'}
        onLoad={() => setIsLoading(false)}
        onError={() => setError(true)}
        className={cn(
          'transition-opacity duration-300',
          isLoading ? 'opacity-0' : 'opacity-100',
          className
        )}
      />
    </picture>
  );
}
```

```tsx
// shared/hooks/useDebounce.ts
import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

// 사용
function SearchInput() {
  const [query, setQuery] = useState('');
  const debouncedQuery = useDebounce(query, 300);

  // debouncedQuery가 변경될 때만 API 호출
  const { data } = useSearch(debouncedQuery);

  return (
    <input
      value={query}
      onChange={(e) => setQuery(e.target.value)}
    />
  );
}

// shared/hooks/useThrottle.ts
import { useRef } from 'react';

export function useThrottle<T extends (...args: any[]) => any>(
  callback: T,
  delay: number
): T {
  const lastCall = useRef(0);

  // React 19 Compiler가 자동 최적화
  const throttled = ((...args) => {
    const now = Date.now();
    if (now - lastCall.current >= delay) {
      lastCall.current = now;
      return callback(...args);
    }
  }) as T;

  return throttled;
}

// 스크롤 이벤트에 사용
function ScrollHandler() {
  const handleScroll = useThrottle(() => {
    console.log('scroll position:', window.scrollY);
  }, 100);

  useEffect(() => {
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, [handleScroll]);
}
```

### LOD (Level of Detail)

```tsx
import { LOD } from 'three';
import { useFrame } from '@react-three/fiber';
import { useRef, useEffect } from 'react';

function LODModel() {
  const lod = useRef<LOD>(null);

  useEffect(() => {
    if (!lod.current) return;

    // 거리별 다른 모델
    lod.current.addLevel(highDetailMesh, 0);     // 가까울 때
    lod.current.addLevel(mediumDetailMesh, 50);  // 중간 거리
    lod.current.addLevel(lowDetailMesh, 100);    // 멀 때
  }, []);

  useFrame(({ camera }) => {
    lod.current?.update(camera);
  });

  return <primitive ref={lod} object={new LOD()} />;
}
```
