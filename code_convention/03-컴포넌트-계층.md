## 계층 개요

```
┌─────────────────────────────────────────────────────────────┐
│                        Page Layer                           │
│  routes/*.tsx - 라우트 진입점, 데이터 로딩, 레이아웃        │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                      Feature Layer                          │
│  features/*/components - 도메인 로직, 비즈니스 컴포넌트     │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                        UI Layer                             │
│  shared/components - 재사용 가능한 UI 컴포넌트              │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                     Primitive Layer                         │
│  shared/components/ui - shadcn/ui 기반 기본 컴포넌트        │
└─────────────────────────────────────────────────────────────┘
```

### 각 계층의 책임

| 계층 | 위치 | 책임 | 예시 |
|------|------|------|------|
| **Page** | `routes/` | 라우트 진입, 데이터 로딩, 레이아웃 조합 | `projects.$id.tsx` |
| **Feature** | `features/*/components/` | 도메인 로직, 비즈니스 규칙 구현 | `ProjectCard`, `LoginForm` |
| **UI** | `shared/components/` | 재사용 UI, 프레젠테이션 로직 | `DataTable`, `Pagination` |
| **Primitive** | `shared/components/ui/` | 기본 UI 요소, 스타일 variants | `Button`, `Input`, `Dialog` |

## Page 컴포넌트

### 역할
- 라우트 진입점
- 서버/클라이언트 데이터 로딩
- Feature 컴포넌트 조합
- 페이지 레이아웃 구성
- SEO 메타데이터 정의

### 구조

```tsx
// routes/projects.$id.tsx
import type { Route } from './+types/projects.$id';
import { ProjectDetail, ProjectSidebar } from '~/features/projects';
import { PageLayout } from '@/components/layout/PageLayout';

// 메타데이터
export function meta({ data }: Route.MetaArgs) {
  const project = data?.project;
  return [
    { title: project?.title ?? 'Project' },
    { name: 'description', content: project?.description },
    { property: 'og:title', content: project?.title },
    { property: 'og:image', content: project?.thumbnail },
  ];
}

// 데이터 로딩
export async function loader({ params }: Route.LoaderArgs) {
  const project = await getProject(params.id);
  if (!project) {
    throw new Response('Not Found', { status: 404 });
  }
  return { project };
}

// 페이지 컴포넌트
export default function ProjectPage({ loaderData }: Route.ComponentProps) {
  const { project } = loaderData;

  return (
    <PageLayout>
      <div className="flex gap-8">
        <main className="flex-1">
          <ProjectDetail project={project} />
        </main>
        <aside className="w-80">
          <ProjectSidebar project={project} />
        </aside>
      </div>
    </PageLayout>
  );
}

// 에러 처리
export function ErrorBoundary({ error }: Route.ErrorBoundaryProps) {
  return (
    <PageLayout>
      <ErrorState
        title="프로젝트를 불러올 수 없습니다"
        message={error.message}
      />
    </PageLayout>
  );
}
```

## Feature 컴포넌트

### 역할
- 도메인/비즈니스 로직 구현
- 상태 관리 (서버 상태, 클라이언트 상태)
- API 호출 및 mutation
- 사용자 인터랙션 처리

### 구조

```tsx
// features/projects/hooks/useProjectFilters.ts
// 공유 훅으로 URL 상태 관리
export function useProjectFilters() {
  return useQueryStates({
    category: parseAsString,
    page: parseAsInteger.withDefault(1),

  return (

        onChange={(category) => setFilters({ category, page: 1 })}

// features/projects/components/ProjectList.tsx
// 리스트 컴포넌트 - 동일한 훅으로 상태 접근
export function ProjectList() {
  const [filters, setFilters] = useProjectFilters();
  const { data } = useProjects(filters); // useSuspenseQuery 사용

  if (data.items.length === 0) {
    return <EmptyState message="프로젝트가 없습니다" />;
  }

  return (
    <>
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {data.items.map((project) => (
          <ProjectCard key={project.id} project={project} />
        ))}
      </div>

      <Pagination
        page={filters.page}
        totalPages={data.totalPages}
        onChange={(page) => setFilters({ page })}
      />
    </>
  );
}

function ProjectListSkeleton() {
  return (
    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
      {Array.from({ length: 6 }).map((_, i) => (
        <Skeleton key={i} className="h-64 rounded-lg" />
      ))}
    </div>
  );
}
```

### Feature 컴포넌트 패턴

#### 훅으로 상태/액션 직접 접근

> **원칙**: Handler를 props로 전달하지 않고, 각 컴포넌트가 필요한 훅을 직접 호출합니다.

```tsx
// features/projects/components/ProjectCard.tsx
// 컴포넌트가 필요한 훅을 직접 사용
export function ProjectCard({ project }: { project: Project }) {
  const navigate = useNavigate();

  return (
    <Card
      onClick={() => navigate(`/projects/${project.id}`)}
      className="cursor-pointer hover:shadow-lg"
    >
      <CardHeader>
        <img src={project.thumbnail} alt={project.title} />
      </CardHeader>
      <CardContent>
        <CardTitle>{project.title}</CardTitle>
        <CardDescription>{project.description}</CardDescription>
      </CardContent>
      <CardFooter>
        <ProjectDeleteButton projectId={project.id} />
      </CardFooter>
    </Card>
  );
}

// 삭제 버튼 - mutation 훅을 직접 사용
function ProjectDeleteButton({ projectId }: { projectId: string }) {
  const { mutate: deleteProject, isPending } = useDeleteProject();

  return (
    <Button
      variant="destructive"
      onClick={(e) => {
        e.stopPropagation();
        deleteProject(projectId);
      }}
      disabled={isPending}
    >
      {isPending ? 'Deleting...' : 'Delete'}
    </Button>
  );
}
```

```tsx
// shared/components/data-display/DataTable.tsx
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@/components/ui/table';
import { Skeleton } from '@/components/ui/skeleton';

type Column<T> = {
  key: keyof T | string;
  header: string;
  render?: (item: T) => React.ReactNode;
  className?: string;
};

type DataTableProps<T> = {
  data: T[];
  columns: Column<T>[];
  isLoading?: boolean;
  emptyMessage?: string;
  onRowClick?: (item: T) => void;
};

export function DataTable<T extends { id: string | number }>({
  data,
  columns,
  isLoading,
  emptyMessage = '데이터가 없습니다',
  onRowClick,
}: DataTableProps<T>) {
  if (isLoading) {
    return <DataTableSkeleton columns={columns.length} />;
  }

  if (data.length === 0) {
    return (
      <div className="text-center py-10 text-muted-foreground">
        {emptyMessage}
      </div>
    );
  }

  return (
    <Table>
      <TableHeader>
        <TableRow>
          {columns.map(column => (
            <TableHead key={String(column.key)} className={column.className}>
              {column.header}
            </TableHead>
          ))}
        </TableRow>
      </TableHeader>
      <TableBody>
        {data.map(item => (
          <TableRow
            key={item.id}
            onClick={() => onRowClick?.(item)}
            className={onRowClick ? 'cursor-pointer hover:bg-muted' : undefined}
          >
            {columns.map(column => (
              <TableCell key={String(column.key)} className={column.className}>
                {column.render
                  ? column.render(item)
                  : String(item[column.key as keyof T] ?? '')}
              </TableCell>
            ))}
          </TableRow>
        ))}
      </TableBody>
    </Table>
  );
}

function DataTableSkeleton({ columns }: { columns: number }) {
  return (
    <div className="space-y-2">
      {Array.from({ length: 5 }).map((_, i) => (
        <Skeleton key={i} className="h-12 w-full" />
      ))}
    </div>
  );
}
```

```tsx
// shared/components/ui/button.tsx
import * as React from 'react';
import { Slot } from '@radix-ui/react-slot';
import { cva, type VariantProps } from 'class-variance-authority';
import { cn } from '@/utils/cn';

const buttonVariants = cva(
  // 기본 스타일
  'inline-flex items-center justify-center whitespace-nowrap rounded-md text-sm font-medium ring-offset-background transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 disabled:pointer-events-none disabled:opacity-50',
  {
    variants: {
      variant: {
        default: 'bg-primary text-primary-foreground hover:bg-primary/90',
        destructive: 'bg-destructive text-destructive-foreground hover:bg-destructive/90',
        outline: 'border border-input bg-background hover:bg-accent hover:text-accent-foreground',
        secondary: 'bg-secondary text-secondary-foreground hover:bg-secondary/80',
        ghost: 'hover:bg-accent hover:text-accent-foreground',
        link: 'text-primary underline-offset-4 hover:underline',
      },
      size: {
        default: 'h-10 px-4 py-2',
        sm: 'h-9 rounded-md px-3',
        lg: 'h-11 rounded-md px-8',
        icon: 'h-10 w-10',
      },
    },
    defaultVariants: {
      variant: 'default',
      size: 'default',
    },
  }
);

export interface ButtonProps
  extends React.ButtonHTMLAttributes<HTMLButtonElement>,
    VariantProps<typeof buttonVariants> {
  asChild?: boolean;
}

const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
  ({ className, variant, size, asChild = false, ...props }, ref) => {
    const Comp = asChild ? Slot : 'button';
    return (
      <Comp
        className={cn(buttonVariants({ variant, size, className }))}
        ref={ref}
        {...props}
      />
    );
  }
);
Button.displayName = 'Button';

export { Button, buttonVariants };
```

### 3. Props 인터페이스 설계

```tsx
// 필수/선택 구분 명확히
type ButtonProps = {
  // 필수
  children: React.ReactNode;

  // 선택 (기본값 있음)
  variant?: 'primary' | 'secondary' | 'ghost';
  size?: 'sm' | 'md' | 'lg';
  disabled?: boolean;

  // 이벤트 (선택)
  onClick?: () => void;

  // 스타일 확장
  className?: string;
};

// HTML 속성 확장
interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label?: string;
  error?: string;
}

// 제네릭 Props
type ListProps<T> = {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  keyExtractor?: (item: T, index: number) => string;
};
```

```tsx
  return (
    <div className="flex items-center gap-2">
      {notification.isNew && <Badge>New</Badge>}
      <span>{notification.message}</span>
      {notification.actionType && (
        <NotificationAction
          notificationId={notification.id}
          actionType={notification.actionType}
        />
      )}
    </div>
  );
}

// 액션 버튼 - 훅으로 직접 mutation 처리
function NotificationAction({
  notificationId,
  actionType,
}: {
  notificationId: string;
  actionType: string;
}) {
  const { mutate } = useNotificationAction();

  return (
    <Button onClick={() => mutate({ notificationId, actionType })}>
      {actionType}
    </Button>
  );
}
```

```tsx
// Render Props가 유용한 경우: 렌더링 커스터마이징
type ListProps<T> = {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  renderEmpty?: () => React.ReactNode;
};

function List<T>({ items, renderItem, renderEmpty }: ListProps<T>) {
  if (items.length === 0) {
    return renderEmpty?.() ?? <EmptyState />;
  }

  return <ul>{items.map((item, i) => renderItem(item, i))}</ul>;
}

// 사용
<List
  items={projects}
  renderItem={(project) => <ProjectCard key={project.id} project={project} />}
  renderEmpty={() => <EmptyState message="프로젝트가 없습니다" />}
/>
```

### HOC (Higher-Order Component)

```tsx
// 인증 래퍼
function withAuth<P extends object>(Component: React.ComponentType<P>) {
  return function AuthenticatedComponent(props: P) {
    const { isAuthenticated, isLoading } = useAuth();

    if (isLoading) return <LoadingSpinner />;
    if (!isAuthenticated) return <Navigate to="/login" />;

    return <Component {...props} />;
  };
}

// 사용
const ProtectedDashboard = withAuth(Dashboard);
```
