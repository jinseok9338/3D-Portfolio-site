## 상태 분류

### 상태 유형별 도구

| 상태 유형 | 도구 | 사용 사례 |
|-----------|------|-----------|
| **URL 상태** | nuqs | 필터, 정렬, 페이지네이션, 검색어, 탭 |
| **서버 상태** | TanStack Query | API 데이터, 캐싱, 동기화 |
| **클라이언트 전역** | Zustand | 인증, 테마, 사이드바 상태 |
| **로컬 상태** | useState/useReducer | 폼 입력, 모달 열림/닫힘 |


### 기본 사용법

```tsx
import { useQueryState, parseAsInteger, parseAsString } from 'nuqs';

function ProjectList() {
  // 문자열 상태 (기본)
  const [search, setSearch] = useQueryState('q');
  // URL: ?q=react

  // 숫자 상태
  const [page, setPage] = useQueryState('page', parseAsInteger.withDefault(1));
  // URL: ?page=2

  // 열거형 상태
  const [sort, setSort] = useQueryState('sort', parseAsString.withDefault('latest'));
  // URL: ?sort=popular

  return (
    <div>
      <input
        value={search ?? ''}
        onChange={(e) => setSearch(e.target.value || null)}
        placeholder="검색..."
      />

      <select
        value={sort}
        onChange={(e) => setSort(e.target.value)}
      >
        <option value="latest">최신순</option>
        <option value="popular">인기순</option>
      </select>

      <Pagination
        page={page}
        onChange={setPage}
      />
    </div>
  );
}
```

```tsx
import {
  parseAsInteger,
  parseAsString,
  parseAsBoolean,
  parseAsArrayOf,
  parseAsStringLiteral,
  createParser,

// 기본 파서
export const parsers = {
  page: parseAsInteger.withDefault(1),
  limit: parseAsInteger.withDefault(20),
  search: parseAsString,
  isActive: parseAsBoolean.withDefault(true),
};

export const sortParser = parseAsStringLiteral(['latest', 'popular', 'oldest'] as const)
  .withDefault('latest');

export const viewModeParser = parseAsStringLiteral(['grid', 'list'] as const)
  .withDefault('grid');

// 배열 파서
export const tagsParser = parseAsArrayOf(parseAsString);
// URL: ?tags=react&tags=typescript

// 커스텀 파서 (날짜)
export const dateParser = createParser({
  parse: (value) => {
    const date = new Date(value);
    return isNaN(date.getTime()) ? null : date;

  serialize: (date) => date.toISOString().split('T')[0],
});
```

### 복합 상태 관리

```tsx
import { useQueryStates, parseAsInteger, parseAsString } from 'nuqs';

// 여러 상태를 하나의 훅으로
function useListFilters() {
  return useQueryStates({
    page: parseAsInteger.withDefault(1),
    limit: parseAsInteger.withDefault(20),
    search: parseAsString,
    sort: parseAsString.withDefault('latest'),
    category: parseAsString,
  });
}

function ProjectList() {
  const [filters, setFilters] = useListFilters();

  // 개별 업데이트
  const handleSearch = (value: string) => {
    setFilters({ search: value || null, page: 1 }); // 검색 시 페이지 리셋
  };

  // 전체 리셋
  const handleReset = () => {
    setFilters({
      page: null,
      limit: null,
      search: null,
      sort: null,
      category: null,
    });
  };

  return (
    <div>
      <input
        value={filters.search ?? ''}
        onChange={(e) => handleSearch(e.target.value)}
      />
      <button onClick={handleReset}>필터 초기화</button>
    </div>
  );
}
```

### Query 훅 패턴 (useSuspenseQuery)

> **권장**: `useQuery` 대신 `useSuspenseQuery`를 사용합니다.
> - `isLoading`, `isError` 체크 불필요
> - `data`는 항상 존재 (undefined 체크 불필요)
> - Suspense + ErrorBoundary로 로딩/에러 UI 분리

```tsx
// features/projects/hooks/useProjects.ts
import { useSuspenseQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { projectKeys } from '../api/projects.queries';
import { fetchProjects, createProject, updateProject, deleteProject } from '../api/projects.api';

// 목록 조회
export function useProjects(filters: ProjectFilters) {
  return useSuspenseQuery({
    queryKey: projectKeys.list(filters),
    queryFn: () => fetchProjects(filters),
  });
}

// 상세 조회
export function useProject(id: string) {
  return useSuspenseQuery({
    queryKey: projectKeys.detail(id),
    queryFn: () => fetchProject(id),
  });
}

// 생성
export function useCreateProject() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: createProject,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: projectKeys.lists() });
    },
  });
}

// 수정
export function useUpdateProject() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: ({ id, data }: { id: string; data: UpdateProjectData }) =>
      updateProject(id, data),
    onSuccess: (_, { id }) => {
      queryClient.invalidateQueries({ queryKey: projectKeys.detail(id) });
      queryClient.invalidateQueries({ queryKey: projectKeys.lists() });
    },
  });
}

// 삭제
export function useDeleteProject() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: deleteProject,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: projectKeys.lists() });
    },
  });
}
```

### Optimistic Updates

```tsx
export function useToggleLike() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: toggleLike,
    onMutate: async (projectId) => {
      // 진행 중인 refetch 취소
      await queryClient.cancelQueries({ queryKey: projectKeys.detail(projectId) });

      // 이전 값 저장
      const previousProject = queryClient.getQueryData(projectKeys.detail(projectId));

      // Optimistic update
      queryClient.setQueryData(
        projectKeys.detail(projectId),
        (old: Project) => ({
          ...old,
          isLiked: !old.isLiked,
          likeCount: old.isLiked ? old.likeCount - 1 : old.likeCount + 1,
        })
      );

      return { previousProject };
    },
    onError: (err, projectId, context) => {
      // 에러 시 롤백
      queryClient.setQueryData(
        projectKeys.detail(projectId),
        context?.previousProject
      );
    },
    onSettled: (_, __, projectId) => {
      // 완료 후 refetch
      queryClient.invalidateQueries({ queryKey: projectKeys.detail(projectId) });
    },
  });
}
```

### Infinite Query

```tsx
import { useInfiniteQuery } from '@tanstack/react-query';

export function useInfiniteProjects(category?: string) {
  return useInfiniteQuery({
    queryKey: ['projects', 'infinite', category],
    queryFn: ({ pageParam = 1 }) => fetchProjects({ page: pageParam, category }),
    getNextPageParam: (lastPage) =>
      lastPage.hasNextPage ? lastPage.page + 1 : undefined,
    initialPageParam: 1,
  });
}

// 사용
function ProjectInfiniteList() {
  const {
    data,
    fetchNextPage,
    hasNextPage,
    isFetchingNextPage,

        page.items.map((project) => (
          <ProjectCard key={project.id} project={project} />

        <button onClick={() => fetchNextPage()} disabled={isFetchingNextPage}>
          {isFetchingNextPage ? '로딩 중...' : '더 보기'}
        </button>
      )}
    </div>
  );
}
```

### 기본 스토어

```tsx
// features/auth/stores/auth.store.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';

type User = {
  id: string;
  email: string;
  name: string;
};

type AuthState = {
  user: User | null;
  accessToken: string | null;
  isAuthenticated: boolean;
};

type AuthActions = {
  login: (user: User, token: string) => void;
  logout: () => void;
  updateUser: (updates: Partial<User>) => void;
};

type AuthStore = AuthState & AuthActions;

export const useAuthStore = create<AuthStore>()(
  persist(
    (set) => ({
      // State
      user: null,
      accessToken: null,
      isAuthenticated: false,

      // Actions
      login: (user, accessToken) =>
        set({ user, accessToken, isAuthenticated: true }),

      logout: () =>
        set({ user: null, accessToken: null, isAuthenticated: false }),

      updateUser: (updates) =>
        set((state) => ({
          user: state.user ? { ...state.user, ...updates } : null,
        })),
    }),
    {
      name: 'auth-storage',
      storage: createJSONStorage(() => localStorage),
      partialize: (state) => ({
        user: state.user,
        accessToken: state.accessToken,
        isAuthenticated: state.isAuthenticated,
      }),
    }
  )
);
```

### useReducer

```tsx
// 복잡한 상태 로직
type FormState = {
  values: Record<string, string>;
  errors: Record<string, string>;
  isSubmitting: boolean;
  isValid: boolean;
};

type FormAction =
  | { type: 'SET_VALUE'; field: string; value: string }
  | { type: 'SET_ERROR'; field: string; error: string }
  | { type: 'CLEAR_ERRORS' }
  | { type: 'START_SUBMIT' }
  | { type: 'END_SUBMIT' }
  | { type: 'RESET' };

function formReducer(state: FormState, action: FormAction): FormState {
  switch (action.type) {
    case 'SET_VALUE':
      return {
        ...state,
        values: { ...state.values, [action.field]: action.value },
      };
    case 'SET_ERROR':
      return {
        ...state,
        errors: { ...state.errors, [action.field]: action.error },
        isValid: false,
      };
    case 'CLEAR_ERRORS':
      return { ...state, errors: {}, isValid: true };
    case 'START_SUBMIT':
      return { ...state, isSubmitting: true };
    case 'END_SUBMIT':
      return { ...state, isSubmitting: false };
    case 'RESET':
      return initialState;
    default:
      return state;
  }
}

function Form() {
  const [state, dispatch] = useReducer(formReducer, initialState);

  const handleChange = (field: string) => (e: React.ChangeEvent<HTMLInputElement>) => {
    dispatch({ type: 'SET_VALUE', field, value: e.target.value });
  };

  // ...
}
```

```tsx
    <>
      <ProjectFilters />
      <ErrorBoundary fallback={<ProjectListError />}>
        <Suspense fallback={<ProjectListSkeleton />}>
          <ProjectListContent />
        </Suspense>
      </ErrorBoundary>
    </>
  );

// 필터 컴포넌트 - 훅으로 직접 상태 접근
function ProjectFilters() {
  const [filters, setFilters] = useProjectFilters();

  return (
    <div>
      <SearchInput
        value={filters.search ?? ''}
        onChange={(value) => setFilters({ search: value || null, page: 1 })}
      />
      <CategorySelect
        value={filters.category}
        onChange={(value) => setFilters({ category: value, page: 1 })}
      />
    </div>
  );
}

// 리스트 컴포넌트 - 동일한 훅으로 상태 접근
function ProjectListContent() {
  const [filters, setFilters] = useProjectFilters();
  const { data } = useProjects(filters);

  return (
    <>
      <ProjectList data={data} />
      <Pagination
        page={filters.page}
        totalPages={data.totalPages}
        onChange={(page) => setFilters({ page })}
      />
    </>
  );
}
```
