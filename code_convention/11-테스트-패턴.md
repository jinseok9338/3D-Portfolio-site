## 테스트 전략

### 테스트 피라미드

```
                 /\
                /  \
               / E2E \          적음 - 느림 - 비용 높음
              /--------\
             /   통합    \
            /--------------\
           /     컴포넌트    \
          /------------------\
         /       단위         \     많음 - 빠름 - 비용 낮음
        /----------------------\
```

### 테스트 범위

| 테스트 유형 | 도구 | 대상 | 빈도 |
|------------|------|------|------|
| **단위** | Vitest | 유틸, 훅, 순수 함수 | 매우 많음 |
| **컴포넌트** | Vitest + Testing Library | UI 컴포넌트 | 많음 |
| **통합** | Vitest + MSW | API 연동, 상태 관리 | 중간 |
| **E2E** | Playwright | 사용자 시나리오 | 핵심 플로우 |

```typescript
// shared/utils/format.test.ts
import { describe, it, expect } from 'vitest';
import { formatDate, formatNumber, formatCurrency } from './format';

describe('formatDate', () => {
  it('날짜를 YYYY-MM-DD 형식으로 포맷한다', () => {
    const date = new Date('2024-01-15');
    expect(formatDate(date)).toBe('2024-01-15');
  });

  it('커스텀 포맷을 적용한다', () => {
    const date = new Date('2024-01-15');
    expect(formatDate(date, 'YYYY년 MM월 DD일')).toBe('2024년 01월 15일');
  });

  it('유효하지 않은 날짜는 빈 문자열을 반환한다', () => {
    expect(formatDate(new Date('invalid'))).toBe('');
  });
});

describe('formatNumber', () => {
  it('천 단위 구분자를 추가한다', () => {
    expect(formatNumber(1234567)).toBe('1,234,567');
  });

  it('소수점을 처리한다', () => {
    expect(formatNumber(1234.56, 2)).toBe('1,234.56');
  });
});

describe('formatCurrency', () => {
  it('원화 형식으로 포맷한다', () => {
    expect(formatCurrency(10000)).toBe('₩10,000');
  });

  it('달러 형식으로 포맷한다', () => {
    expect(formatCurrency(100, 'USD')).toBe('$100.00');
  });
});
```

```typescript
// shared/hooks/useDebounce.test.ts
import { describe, it, expect, vi } from 'vitest';
import { renderHook, act, waitFor } from '@testing-library/react';
import { useDebounce } from './useDebounce';

describe('useDebounce', () => {
  beforeEach(() => {
    vi.useFakeTimers();
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it('초기값을 즉시 반환한다', () => {
    const { result } = renderHook(() => useDebounce('initial', 500));
    expect(result.current).toBe('initial');
  });

  it('지연 시간 후 값을 업데이트한다', async () => {
    const { result, rerender } = renderHook(
      ({ value }) => useDebounce(value, 500),
      { initialProps: { value: 'initial' } }
    );

    rerender({ value: 'updated' });
    expect(result.current).toBe('initial');

    act(() => {
      vi.advanceTimersByTime(500);
    });

    expect(result.current).toBe('updated');
  });

  it('연속 변경 시 마지막 값만 반영한다', async () => {
    const { result, rerender } = renderHook(
      ({ value }) => useDebounce(value, 500),
      { initialProps: { value: 'a' } }
    );

    rerender({ value: 'b' });
    act(() => vi.advanceTimersByTime(200));

    rerender({ value: 'c' });
    act(() => vi.advanceTimersByTime(200));

    rerender({ value: 'd' });
    act(() => vi.advanceTimersByTime(500));

    expect(result.current).toBe('d');
  });
});
```

```typescript
// features/auth/stores/auth.store.test.ts
import { describe, it, expect, beforeEach } from 'vitest';
import { useAuthStore } from './auth.store';

describe('useAuthStore', () => {
  beforeEach(() => {
    // 스토어 초기화
    useAuthStore.setState({
      user: null,
      accessToken: null,
      isAuthenticated: false,
    });
  });

  it('login 시 유저와 토큰을 저장한다', () => {
    const user = { id: '1', email: 'test@example.com', name: 'Test' };
    const token = 'test-token';

    useAuthStore.getState().login(user, token);

    const state = useAuthStore.getState();
    expect(state.user).toEqual(user);
    expect(state.accessToken).toBe(token);
    expect(state.isAuthenticated).toBe(true);
  });

  it('logout 시 상태를 초기화한다', () => {
    // 먼저 로그인
    useAuthStore.getState().login(
      { id: '1', email: 'test@example.com', name: 'Test' },
      'token'
    );

    useAuthStore.getState().logout();

    const state = useAuthStore.getState();
    expect(state.user).toBeNull();
    expect(state.accessToken).toBeNull();
    expect(state.isAuthenticated).toBe(false);
  });
});
```

```tsx
    await waitFor(() => {
      expect(screen.getByText(/이메일을 입력해주세요/i)).toBeInTheDocument();
    });
  });

  it('이메일 형식 검사를 수행한다', async () => {
    const user = userEvent.setup();
    renderWithProviders(<LoginForm />);

    await user.type(screen.getByLabelText(/이메일/i), 'invalid-email');
    await user.click(screen.getByRole('button', { name: /로그인/i }));

    await waitFor(() => {
      expect(screen.getByText(/올바른 이메일을 입력해주세요/i)).toBeInTheDocument();
    });
  });

  it('올바른 입력 시 제출한다', async () => {

    const onSubmit = vi.fn();

        email: 'test@example.com',
        password: 'password123',
      });
    });
  });
```

```typescript
export const handlers = [

      data: [

      ],
      pagination: { page: 1, totalPages: 1, total: 2 },
    });
  }),

  // 프로젝트 상세

    return HttpResponse.json({
      id: params.id,
      title: `Project ${params.id}`,
      description: 'Detailed description',
    });
  }),

  // 프로젝트 생성

    const body = await request.json();

      id: 'new-id',
```

### TanStack Query 통합 테스트

```tsx
// features/projects/hooks/useProjects.test.tsx
import { describe, it, expect } from 'vitest';
import { renderHook, waitFor } from '@testing-library/react';
import { useProjects, useCreateProject } from './useProjects';
import { createWrapper } from '@/test/utils';
import { server } from '@/test/mocks/server';
import { http, HttpResponse } from 'msw';

describe('useProjects', () => {
  it('프로젝트 목록을 조회한다', async () => {
    const { result } = renderHook(() => useProjects(), {
      wrapper: createWrapper(),
    });

    expect(result.current.isLoading).toBe(true);

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    expect(result.current.data?.data).toHaveLength(2);
  });

  it('필터를 적용한다', async () => {
    server.use(
      http.get('/api/projects', ({ request }) => {
        const url = new URL(request.url);
        const category = url.searchParams.get('category');

        const projects = [
          { id: '1', title: 'Web Project', category: 'web' },
          { id: '2', title: 'Mobile Project', category: 'mobile' },
        ].filter((p) => !category || p.category === category);

        return HttpResponse.json({ data: projects, pagination: {} });
      })
    );

    const { result } = renderHook(
      () => useProjects({ category: 'web' }),
      { wrapper: createWrapper() }
    );

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    expect(result.current.data?.data).toHaveLength(1);
    expect(result.current.data?.data[0].category).toBe('web');
  });
});

describe('useCreateProject', () => {
  it('프로젝트를 생성한다', async () => {
    const { result } = renderHook(() => useCreateProject(), {
      wrapper: createWrapper(),
    });

    result.current.mutate({
      title: 'New Project',
      description: 'Description',
      category: 'web',
    });

    await waitFor(() => {
      expect(result.current.isSuccess).toBe(true);
    });

    expect(result.current.data?.title).toBe('New Project');
  });
});
```

```typescript
import { test, expect } from '@playwright/test';

test.describe('Project CRUD', () => {
  test.beforeEach(async ({ page }) => {
    // 인증된 상태로 시작
    await page.goto('/login');
    await page.getByLabel('이메일').fill('test@example.com');
    await page.getByLabel('비밀번호').fill('password123');
    await page.getByRole('button', { name: '로그인' }).click();
    await page.waitForURL('/dashboard');
  });

  test('프로젝트 생성', async ({ page }) => {
    await page.goto('/projects');
    await page.getByRole('button', { name: '새 프로젝트' }).click();

    // 폼 작성
    await page.getByLabel('프로젝트 이름').fill('E2E Test Project');
    await page.getByLabel('설명').fill('This is a test project created by E2E test');
    await page.getByLabel('카테고리').selectOption('web');

    // 제출
    await page.getByRole('button', { name: '생성' }).click();

    // 성공 확인
    await expect(page.getByText('프로젝트가 생성되었습니다')).toBeVisible();
    await expect(page.getByText('E2E Test Project')).toBeVisible();
  });

  test('프로젝트 수정', async ({ page }) => {
    await page.goto('/projects/1');
    await page.getByRole('button', { name: '수정' }).click();

    await page.getByLabel('프로젝트 이름').clear();
    await page.getByLabel('프로젝트 이름').fill('Updated Project Name');
    await page.getByRole('button', { name: '저장' }).click();

    await expect(page.getByText('Updated Project Name')).toBeVisible();
  });

  test('프로젝트 삭제', async ({ page }) => {
    await page.goto('/projects/1');
    await page.getByRole('button', { name: '삭제' }).click();

    // 확인 다이얼로그
    await page.getByRole('button', { name: '삭제 확인' }).click();

    await expect(page).toHaveURL('/projects');
    await expect(page.getByText('Project 1')).not.toBeVisible();
  });
});
```

### 렌더 유틸리티

```tsx
// src/test/utils.tsx
import { ReactElement } from 'react';
import { render, RenderOptions } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { MemoryRouter } from 'react-router';

// Query Client 생성
function createTestQueryClient() {
  return new QueryClient({
    defaultOptions: {
      queries: {
        retry: false,
        gcTime: 0,
      },
    },
  });
}

// Provider 래퍼
function AllProviders({ children }: { children: React.ReactNode }) {
  const queryClient = createTestQueryClient();

  return (
    <QueryClientProvider client={queryClient}>
      <MemoryRouter>
        {children}
      </MemoryRouter>
    </QueryClientProvider>
  );
}

// 커스텀 렌더
function renderWithProviders(
  ui: ReactElement,
  options?: Omit<RenderOptions, 'wrapper'>
) {
  return render(ui, { wrapper: AllProviders, ...options });
}

// 훅 래퍼
function createWrapper() {
  const queryClient = createTestQueryClient();
  return ({ children }: { children: React.ReactNode }) => (
    <QueryClientProvider client={queryClient}>
      {children}
    </QueryClientProvider>
  );
}

export { renderWithProviders, createWrapper };
```

```typescript
// src/test/factories/project.ts
import { faker } from '@faker-js/faker';
import type { Project } from '@/features/projects';

export function createProject(overrides: Partial<Project> = {}): Project {
  return {
    id: faker.string.uuid(),
    title: faker.commerce.productName(),
    description: faker.lorem.paragraph(),
    category: faker.helpers.arrayElement(['web', 'mobile', 'design']),
    thumbnail: faker.image.url(),
    technologies: faker.helpers.arrayElements(['React', 'TypeScript', 'Node.js'], 3),
    githubUrl: faker.internet.url(),
    liveUrl: faker.internet.url(),
    featured: faker.datatype.boolean(),
    createdAt: faker.date.past().toISOString(),
    updatedAt: faker.date.recent().toISOString(),
    ...overrides,
  };
}

export function createProjects(count: number): Project[] {
  return Array.from({ length: count }, () => createProject());
}
```
