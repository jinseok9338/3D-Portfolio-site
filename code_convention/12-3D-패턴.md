## R3F 개요

### 핵심 라이브러리

```bash
npm install three @react-three/fiber @react-three/drei @react-three/postprocessing
npm install -D @types/three
```

| 라이브러리 | 용도 |
|-----------|------|
| `three` | Three.js 코어 |
| `@react-three/fiber` | React 렌더러 |
| `@react-three/drei` | 유틸리티 컴포넌트 |
| `@react-three/postprocessing` | 후처리 효과 |

### 기본 개념

```tsx
import { Canvas } from '@react-three/fiber';

// Canvas = Three.js Scene + WebGL Renderer
// mesh = 3D 객체 (geometry + material)
// primitive = Three.js 객체 직접 사용

function Scene() {
  return (
    <Canvas>
      {/* 조명 */}
      <ambientLight intensity={0.5} />
      <directionalLight position={[10, 10, 5]} />

      {/* 3D 객체 */}
      <mesh>
        <boxGeometry args={[1, 1, 1]} />
        <meshStandardMaterial color="orange" />
      </mesh>

      {/* 카메라 컨트롤 */}
      <OrbitControls />
    </Canvas>
  );
}
```

```tsx
import { Canvas, type CanvasProps } from '@react-three/fiber';
import { Preload, AdaptiveDpr, AdaptiveEvents } from '@react-three/drei';
import { Suspense } from 'react';

type Canvas3DProps = CanvasProps & {
  children: React.ReactNode;
  fallback?: React.ReactNode;
};

export function Canvas3D({
  children,
  fallback = <CanvasLoader />,
  ...props
}: Canvas3DProps) {
  return (
    <Canvas
      dpr={[1, 2]}
      gl={{
        antialias: true,
        alpha: true,
        powerPreference: 'high-performance',
      }}
      camera={{
        fov: 45,
        near: 0.1,
        far: 100,
        position: [0, 0, 5],
      }}
      {...props}
    >
      <Suspense fallback={fallback}>
        {/* 성능 최적화 */}
        <AdaptiveDpr pixelated />
        <AdaptiveEvents />

        {children}

        {/* 에셋 프리로드 */}
        <Preload all />
      </Suspense>
    </Canvas>
  );
}

// 로딩 표시 (HTML 오버레이)

      </div>
```

```tsx
// features/3d/components/Lights.tsx
import { useHelper } from '@react-three/drei';
import { useRef } from 'react';
import * as THREE from 'three';

type LightsProps = {
  debug?: boolean;
};

export function Lights({ debug = false }: LightsProps) {
  const directionalRef = useRef<THREE.DirectionalLight>(null);

  // 디버그 모드에서 조명 헬퍼 표시
  useHelper(
    debug && directionalRef,
    THREE.DirectionalLightHelper,
    1,
    'red'
  );

  return (
    <>
      {/* 환경광 */}
      <ambientLight intensity={0.4} />

      {/* 주 조명 */}
      <directionalLight
        ref={directionalRef}
        position={[5, 5, 5]}
        intensity={1}
        castShadow
        shadow-mapSize={[1024, 1024]}
      />

      {/* 보조 조명 */}
      <directionalLight
        position={[-5, 5, -5]}
        intensity={0.5}
      />

      {/* 림 라이트 */}
      <pointLight
        position={[0, 5, -5]}
        intensity={0.5}
        color="#ffffff"
      />
    </>
  );
}
```

## 컴포넌트 패턴

### 기본 메시 컴포넌트

```tsx
// features/3d/components/Box.tsx
import { useRef, useState } from 'react';
import { useFrame } from '@react-three/fiber';
import { type Mesh } from 'three';

type BoxProps = {
  position?: [number, number, number];
  color?: string;
  size?: number;
  animate?: boolean;
};

export function Box({
  position = [0, 0, 0],
  color = '#ff6b6b',
  size = 1,
  animate = true,
}: BoxProps) {
  const meshRef = useRef<Mesh>(null);
  const [hovered, setHovered] = useState(false);
  const [clicked, setClicked] = useState(false);

  useFrame((state, delta) => {
    if (!animate || !meshRef.current) return;
    meshRef.current.rotation.x += delta * 0.5;
    meshRef.current.rotation.y += delta * 0.5;
  });

  return (
    <mesh
      ref={meshRef}
      position={position}
      scale={clicked ? 1.5 : 1}
      onClick={() => setClicked(!clicked)}
      onPointerOver={() => setHovered(true)}
      onPointerOut={() => setHovered(false)}
    >
      <boxGeometry args={[size, size, size]} />
      <meshStandardMaterial color={hovered ? '#ffa8a8' : color} />
    </mesh>
  );
}
```

### GLTF 모델 컴포넌트

```tsx
// features/3d/components/models/Character.tsx
import { useRef } from 'react';
import { useGLTF, useAnimations } from '@react-three/drei';
import { useFrame } from '@react-three/fiber';
import { type Group } from 'three';

type CharacterProps = {
  position?: [number, number, number];
  rotation?: [number, number, number];
  scale?: number;
  animation?: string;
};

export function Character({
  position = [0, 0, 0],
  rotation = [0, 0, 0],
  scale = 1,
  animation = 'idle',
}: CharacterProps) {
  const groupRef = useRef<Group>(null);
  const { scene, animations } = useGLTF('/models/character.glb');
  const { actions, names } = useAnimations(animations, groupRef);

  // 애니메이션 적용
  useEffect(() => {
    const action = actions[animation];
    if (action) {
      action.reset().fadeIn(0.5).play();
      return () => {
        action.fadeOut(0.5);
      };
    }
  }, [animation, actions]);

  return (
    <group ref={groupRef} position={position} rotation={rotation} scale={scale}>
      <primitive object={scene.clone()} />
    </group>
  );
}

// 프리로드
useGLTF.preload('/models/character.glb');
```

```tsx
function RotatingMesh() {
  const meshRef = useRef<Mesh>(null);

  useFrame((state, delta) => {
    if (!meshRef.current) return;
    meshRef.current.rotation.y += delta;
  });

  return (
    <mesh ref={meshRef}>
      <boxGeometry />
      <meshStandardMaterial />
    </mesh>
  );
}

// 사인파 애니메이션
function FloatingMesh() {
  const meshRef = useRef<Mesh>(null);

  useFrame((state) => {
    if (!meshRef.current) return;
    meshRef.current.position.y = Math.sin(state.clock.elapsedTime) * 0.5;
  });

  return (
    <mesh ref={meshRef}>
      <sphereGeometry args={[0.5, 32, 32]} />
      <meshStandardMaterial />
    </mesh>
  );
}
```

### 스크롤 연동 애니메이션

```tsx
// features/3d/hooks/useScrollAnimation.ts
import { useScroll } from '@react-three/drei';
import { useFrame } from '@react-three/fiber';
import { useRef } from 'react';
import { type Group } from 'three';
import { MathUtils } from 'three';

export function useScrollAnimation(config: {
  startOffset?: number;
  endOffset?: number;
}) {
  const groupRef = useRef<Group>(null);
  const scroll = useScroll();

  useFrame(() => {
    if (!groupRef.current) return;

    const { startOffset = 0, endOffset = 1 } = config;
    const range = scroll.range(startOffset, endOffset);

    // 스크롤에 따른 애니메이션
    groupRef.current.rotation.y = MathUtils.lerp(0, Math.PI * 2, range);
    groupRef.current.position.y = MathUtils.lerp(0, 5, range);
  });

  return groupRef;
}

// 사용
function ScrollAnimatedModel() {
  const groupRef = useScrollAnimation({ startOffset: 0, endOffset: 0.5 });

  return (
    <group ref={groupRef}>
      <Character />
    </group>
  );
}
```

```tsx
// features/3d/hooks/usePerformance.ts
import { useThree } from '@react-three/fiber';

export function usePerformanceLevel() {
  const { gl } = useThree();

  const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
  const isLowEnd = gl.capabilities.maxTextureSize < 4096;

  if (isMobile || isLowEnd) return 'low';
  if (gl.capabilities.maxTextureSize >= 16384) return 'high';
  return 'medium';
}

// 사용
function AdaptiveScene() {
  const level = usePerformanceLevel();

  return (
    <>
      {level === 'high' && <HighQualityEffects />}
      {level === 'medium' && <MediumQualityEffects />}
      {level === 'low' && <LowQualityEffects />}

      <Model quality={level} />
    </>
  );
}
```

### 메모리 관리

```tsx
// 정리 함수
import { useEffect } from 'react';
import { useThree } from '@react-three/fiber';

function MemoryManagedScene() {
  const { gl, scene } = useThree();

  useEffect(() => {
    return () => {
      // 텍스처 정리
      scene.traverse((object) => {
        if (object instanceof Mesh) {
          object.geometry?.dispose();
          if (object.material instanceof Material) {
            object.material.dispose();
          }
        }
      });
    };
  }, [scene]);

  return <Scene />;
}

// GLTF 모델 정리
import { dispose } from '@react-three/fiber';

useEffect(() => {
  return () => dispose(scene);
}, [scene]);
```

## 인터랙션

### 마우스 인터랙션

```tsx
import { useRef, useState } from 'react';
import { ThreeEvent } from '@react-three/fiber';

function InteractiveMesh() {
  const [hovered, setHovered] = useState(false);
  const [active, setActive] = useState(false);

  const handlePointerOver = (e: ThreeEvent<PointerEvent>) => {
    e.stopPropagation();
    setHovered(true);
    document.body.style.cursor = 'pointer';
  };

  const handlePointerOut = () => {
    setHovered(false);
    document.body.style.cursor = 'auto';
  };

  const handleClick = (e: ThreeEvent<MouseEvent>) => {
    e.stopPropagation();
    setActive(!active);
  };

  return (
    <mesh
      scale={active ? 1.5 : 1}
      onPointerOver={handlePointerOver}
      onPointerOut={handlePointerOut}
      onClick={handleClick}
    >
      <boxGeometry />
      <meshStandardMaterial color={hovered ? 'hotpink' : 'orange'} />
    </mesh>
  );
}
```

```tsx
import { useSpring, animated } from '@react-spring/three';

function DraggableMesh() {
  const [spring, api] = useSpring(() => ({
    position: [0, 0, 0],
    config: { mass: 1, tension: 200, friction: 20 },
  }));

  const bind = useDrag(({ offset: [x, y], down }) => {
    api.start({
      position: [x / 100, -y / 100, down ? 1 : 0],
    });
  });

  return (
    <animated.mesh {...spring} {...(bind() as any)}>
      <boxGeometry />
      <meshStandardMaterial color="royalblue" />
    </animated.mesh>
  );
}
```
